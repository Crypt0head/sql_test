Superset graphs SQL (presentation layout)
Source table: sbx_da.tmp_shestopalov_acq_datamart_q1
Client activity rule in this file: active if active_terms > 0

How to use:
1) Copy needed query into SQL Lab and save as virtual dataset, or use logic in Custom SQL metrics.
2) Replace {{month_start}} with first day of selected month (example: 2026-06-01) if you run in SQL Lab.
3) In dashboard mode, month filtering should be done by snapshot_month_start.
4) Prefer snapshot_month_start filter. Do not rely on report_month IN (...) as primary month filter.

1) Clients block
------------------------------------------------------------------------------
WITH base AS (
    SELECT
        CAST(snapshot_month_start AS DATE) AS snapshot_month_start_dt,
        NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') AS agr_key,
        COALESCE(CAST(NULLIF(BTRIM(CAST(active_terms AS TEXT)), '') AS NUMERIC), 0) AS active_terms_num,
        CAST(NULLIF(BTRIM(CAST(d_valid_from AS TEXT)), '') AS DATE) AS d_valid_from_dt,
        CAST(NULLIF(BTRIM(CAST(d_valid_to AS TEXT)), '') AS DATE) AS d_valid_to_dt
    FROM sbx_da.tmp_shestopalov_acq_datamart_q1
),
scope AS (
    SELECT *
    FROM base
    WHERE date_trunc('month', snapshot_month_start_dt) = date_trunc('month', DATE '{{month_start}}')
)
SELECT
    COUNT(DISTINCT CASE WHEN agr_key IS NOT NULL THEN agr_key END) AS clients_total,
    COUNT(DISTINCT CASE WHEN agr_key IS NOT NULL AND active_terms_num > 0 THEN agr_key END) AS clients_active,
    COUNT(DISTINCT CASE WHEN agr_key IS NOT NULL THEN agr_key END)
      - COUNT(DISTINCT CASE WHEN agr_key IS NOT NULL AND active_terms_num > 0 THEN agr_key END) AS clients_passive,
    COUNT(DISTINCT CASE WHEN agr_key IS NOT NULL AND date_trunc('month', d_valid_from_dt) = date_trunc('month', DATE '{{month_start}}') THEN agr_key END) AS clients_new_contracts,
    COUNT(DISTINCT CASE WHEN agr_key IS NOT NULL AND d_valid_to_dt IS NOT NULL AND date_trunc('month', d_valid_to_dt) = date_trunc('month', DATE '{{month_start}}') THEN agr_key END) AS clients_terminated_contracts
FROM scope;

WITH base AS (
    SELECT
        CAST(snapshot_month_start AS DATE) AS snapshot_month_start_dt,
        COALESCE(CAST(NULLIF(BTRIM(CAST(retl_cnt AS TEXT)), '') AS NUMERIC), 0) AS retl_cnt_num,
        COALESCE(CAST(NULLIF(BTRIM(CAST(term_cnt AS TEXT)), '') AS NUMERIC), 0) AS term_cnt_num,
        COALESCE(CAST(NULLIF(BTRIM(CAST(active_terms AS TEXT)), '') AS NUMERIC), 0) AS active_terms_num
    FROM sbx_da.tmp_shestopalov_acq_datamart_q1
),
scope AS (
    SELECT *
    FROM base
    WHERE date_trunc('month', snapshot_month_start_dt) = date_trunc('month', DATE '{{month_start}}')
)
SELECT
    SUM(retl_cnt_num) AS merchants_total,
    SUM(CASE WHEN active_terms_num > 0 THEN retl_cnt_num ELSE 0 END) AS merchants_active_proxy,
    SUM(CASE WHEN active_terms_num <= 0 THEN retl_cnt_num ELSE 0 END) AS merchants_passive_proxy,
    NULL::NUMERIC AS merchants_closed_in_period_na,
    SUM(term_cnt_num) AS terminals_total,
    SUM(active_terms_num) AS terminals_active,
    SUM(term_cnt_num) - SUM(active_terms_num) AS terminals_passive,
    NULL::NUMERIC AS terminals_disconnected_in_period_na
FROM scope;

3) CHOD block (counts + sums)
------------------------------------------------------------------------------
WITH base AS (
    SELECT
        CAST(snapshot_month_start AS DATE) AS snapshot_month_start_dt,
        NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') AS agr_key,
        COALESCE(CAST(NULLIF(BTRIM(CAST(chod AS TEXT)), '') AS NUMERIC), 0) AS chod_num
    FROM sbx_da.tmp_shestopalov_acq_datamart_q1
),
scope AS (
    SELECT *
    FROM base
    WHERE date_trunc('month', snapshot_month_start_dt) = date_trunc('month', DATE '{{month_start}}')
),
bucketed AS (
    SELECT
        CASE
            WHEN chod_num > 0 THEN 'Клиенты с положительным ЧОД ТЭ'
            WHEN chod_num < 0 THEN 'Клиенты с отрицательным ЧОД ТЭ'
            ELSE 'Клиенты с 0 ЧОД ТЭ'
        END AS chod_bucket,
        agr_key,
        chod_num
    FROM scope
    WHERE agr_key IS NOT NULL
)
SELECT
    chod_bucket,
    COUNT(DISTINCT agr_key) AS clients_cnt,
    ROUND(SUM(chod_num) / 1000000.0, 4) AS chod_sum_mln_rub,
    COUNT(DISTINCT CASE WHEN chod_num < 0 THEN agr_key END) AS neg_chod_clients_cnt
FROM bucketed
GROUP BY chod_bucket
ORDER BY chod_bucket;

4) One-line custom metrics (for Big Number charts)
------------------------------------------------------------------------------
clients_total:
COUNT(DISTINCT CASE WHEN NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') IS NOT NULL THEN NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') END)

clients_active:
COUNT(DISTINCT CASE WHEN NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') IS NOT NULL AND COALESCE(CAST(NULLIF(BTRIM(CAST(active_terms AS TEXT)), '') AS NUMERIC), 0) > 0 THEN NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') END)

clients_passive:
COUNT(DISTINCT CASE WHEN NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') IS NOT NULL THEN NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') END)
-
COUNT(DISTINCT CASE WHEN NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') IS NOT NULL AND COALESCE(CAST(NULLIF(BTRIM(CAST(active_terms AS TEXT)), '') AS NUMERIC), 0) > 0 THEN NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') END)

clients_new_contracts:
COUNT(DISTINCT CASE WHEN NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') IS NOT NULL AND d_valid_from IS NOT NULL AND date_trunc('month', CAST(NULLIF(BTRIM(CAST(d_valid_from AS TEXT)), '') AS DATE)) = date_trunc('month', CAST(snapshot_month_start AS DATE)) THEN NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') END)

clients_terminated_contracts:
COUNT(DISTINCT CASE WHEN NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') IS NOT NULL AND d_valid_to IS NOT NULL AND date_trunc('month', CAST(NULLIF(BTRIM(CAST(d_valid_to AS TEXT)), '') AS DATE)) = date_trunc('month', CAST(snapshot_month_start AS DATE)) THEN NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') END)

retail_points_total:
SUM(COALESCE(CAST(NULLIF(BTRIM(CAST(retl_cnt AS TEXT)), '') AS NUMERIC), 0))

retail_points_active_minus_10pct:
SUM(COALESCE(CAST(NULLIF(BTRIM(CAST(active_terms AS TEXT)), '') AS NUMERIC), 0)) * 0.9

retail_points_passive:
SUM(COALESCE(CAST(NULLIF(BTRIM(CAST(retl_cnt AS TEXT)), '') AS NUMERIC), 0))
-
(
  SUM(COALESCE(CAST(NULLIF(BTRIM(CAST(active_terms AS TEXT)), '') AS NUMERIC), 0)) * 0.9
)

terminals_total:
SUM(COALESCE(CAST(NULLIF(BTRIM(CAST(term_cnt AS TEXT)), '') AS NUMERIC), 0))

terminals_active:
SUM(COALESCE(CAST(NULLIF(BTRIM(CAST(active_terms AS TEXT)), '') AS NUMERIC), 0))

terminals_inactive:
SUM(COALESCE(CAST(NULLIF(BTRIM(CAST(term_cnt AS TEXT)), '') AS NUMERIC), 0))
-
SUM(COALESCE(CAST(NULLIF(BTRIM(CAST(active_terms AS TEXT)), '') AS NUMERIC), 0))

chod_sum_total:
SUM(COALESCE(CAST(NULLIF(BTRIM(CAST(chod AS TEXT)), '') AS NUMERIC), 0))

5) Virtual dataset for 3x4 CHOD table (single-month dashboard mode)
------------------------------------------------------------------------------
Dataset name (recommended): vd_chod_clients_by_month

SQL for virtual dataset:

WITH base AS (
    SELECT
        CAST(snapshot_month_start AS DATE) AS snapshot_month_start,
        NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') AS agr_id_key,
        COALESCE(CAST(NULLIF(BTRIM(CAST(chod AS TEXT)), '') AS NUMERIC), 0) AS chod_num
    FROM sbx_da.tmp_shestopalov_acq_datamart_q1
    WHERE NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') IS NOT NULL
),
by_client_month AS (
    SELECT
        snapshot_month_start,
        agr_id_key,
        SUM(chod_num) AS chod_sum
    FROM base
    GROUP BY snapshot_month_start, agr_id_key
)
SELECT
    snapshot_month_start,
    CASE
        WHEN chod_sum > 0 THEN 'Клиенты с положительным ЧОД ТЭ'
        WHEN chod_sum < 0 THEN 'Клиенты с отрицательным ЧОД ТЭ'
        ELSE 'Клиенты с 0 ЧОД ТЭ'
    END AS client_bucket,
    agr_id_key,
    chod_sum,
    CAST(NULL AS NUMERIC) AS total_chod_placeholder
FROM by_client_month;

8) Sanity-check query (clients and CHOD convergence)
------------------------------------------------------------------------------
Replace {{month_start}} with first day of month (example: 2026-04-01)

WITH base AS (
    SELECT
        CAST(snapshot_month_start AS DATE) AS snapshot_month_start,
        NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') AS agr_id_key,
        COALESCE(CAST(NULLIF(BTRIM(CAST(chod AS TEXT)), '') AS NUMERIC), 0) AS chod_num
    FROM sbx_da.tmp_shestopalov_acq_datamart_q1
    WHERE NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') IS NOT NULL
      AND date_trunc('month', CAST(snapshot_month_start AS DATE)) = date_trunc('month', DATE '{{month_start}}')
),
by_client_month AS (
    SELECT
        snapshot_month_start,
        agr_id_key,
        SUM(chod_num) AS chod_sum
    FROM base
    GROUP BY snapshot_month_start, agr_id_key
),
bucketed AS (
    SELECT
        CASE
            WHEN chod_sum > 0 THEN 'pos'
            WHEN chod_sum < 0 THEN 'neg'
            ELSE 'zero'
        END AS bucket,
        agr_id_key,
        chod_sum
    FROM by_client_month
)
SELECT
    (SELECT COUNT(*) FROM by_client_month) AS expected_clients_total,
    (SELECT COALESCE(SUM(chod_sum), 0) FROM by_client_month) AS expected_chod_total,
    (SELECT COUNT(*) FROM bucketed) AS actual_clients_total_from_buckets,
    (SELECT COALESCE(SUM(chod_sum), 0) FROM bucketed) AS actual_chod_total_from_buckets,
    (SELECT SUM(CASE WHEN bucket = 'pos' THEN 1 ELSE 0 END) FROM bucketed) AS pos_clients,
    (SELECT SUM(CASE WHEN bucket = 'neg' THEN 1 ELSE 0 END) FROM bucketed) AS neg_clients,
    (SELECT SUM(CASE WHEN bucket = 'zero' THEN 1 ELSE 0 END) FROM bucketed) AS zero_clients;

9) Unified virtual dataset for 3 charts (Month <= selected Month)
------------------------------------------------------------------------------
Dataset name (recommended): vd_month_upto_selected_metrics

SQL for virtual dataset:

WITH raw AS (
    SELECT
        NULLIF(SUBSTRING(BTRIM(CAST(report_month AS TEXT)) FROM 1 FOR 7), '') AS point_report_month,
        NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') AS agr_id_key,
        COALESCE(CAST(NULLIF(BTRIM(CAST(active_terms AS TEXT)), '') AS NUMERIC), 0) AS active_terms_num,
        COALESCE(CAST(NULLIF(BTRIM(CAST(chod AS TEXT)), '') AS NUMERIC), 0) AS chod_num,
        COALESCE(CAST(NULLIF(BTRIM(CAST(trx_sum AS TEXT)), '') AS NUMERIC), 0) AS trx_sum_num
    FROM sbx_da.tmp_shestopalov_acq_datamart_q1
    WHERE NULLIF(BTRIM(CAST(agr_id AS TEXT)), '') IS NOT NULL
      AND NULLIF(BTRIM(CAST(report_month AS TEXT)), '') IS NOT NULL
),
by_client_month AS (
    SELECT
        point_report_month,
        agr_id_key,
        MAX(CASE WHEN active_terms_num > 0 THEN 1 ELSE 0 END) AS is_active_client,
        SUM(chod_num) AS chod_sum_client_month
    FROM raw
    WHERE point_report_month IS NOT NULL
    GROUP BY point_report_month, agr_id_key
),
monthly_clients AS (
    SELECT
        point_report_month,
        COUNT(*) AS total_clients_cnt,
        SUM(is_active_client) AS active_clients_cnt,
        COUNT(*) - SUM(is_active_client) AS passive_clients_cnt,
        SUM(CASE WHEN chod_sum_client_month > 0 AND chod_sum_client_month <= 2500 THEN 1 ELSE 0 END) AS cohort_0_2500_cnt,
        SUM(CASE WHEN chod_sum_client_month > 2500 THEN 1 ELSE 0 END) AS cohort_gt_2500_cnt,
        SUM(CASE WHEN chod_sum_client_month < 0 THEN 1 ELSE 0 END) AS cohort_lt_0_cnt,
        SUM(CASE WHEN chod_sum_client_month = 0 THEN 1 ELSE 0 END) AS cohort_eq_0_cnt
    FROM by_client_month
    GROUP BY point_report_month
),
monthly_trx AS (
    SELECT
        point_report_month,
        SUM(trx_sum_num) AS trx_sum_total
    FROM raw
    WHERE point_report_month IS NOT NULL
    GROUP BY point_report_month
),
monthly_metrics AS (
    SELECT
        c.point_report_month,
        c.active_clients_cnt,
        c.passive_clients_cnt,
        c.cohort_0_2500_cnt,
        c.cohort_gt_2500_cnt,
        c.cohort_lt_0_cnt,
        c.cohort_eq_0_cnt,
        t.trx_sum_total
    FROM monthly_clients c
    LEFT JOIN monthly_trx t
      ON t.point_report_month = c.point_report_month
),
selected_months AS (
    SELECT DISTINCT point_report_month AS report_month
    FROM monthly_metrics
)
SELECT
    s.report_month,
    m.point_report_month,
    m.active_clients_cnt,
    m.passive_clients_cnt,
    m.cohort_0_2500_cnt,
    m.cohort_gt_2500_cnt,
    m.cohort_lt_0_cnt,
    m.cohort_eq_0_cnt,
    m.trx_sum_total
FROM selected_months s
JOIN monthly_metrics m
  ON m.point_report_month <= s.report_month
ORDER BY s.report_month, m.point_report_month;

14) Sanity-check SQL
------------------------------------------------------------------------------
Replace '2026-05' with selected month.

A) Expansion boundary check (must show period up to selected month):

SELECT
    report_month,
    MIN(point_report_month) AS min_point_report_month,
    MAX(point_report_month) AS max_point_report_month,
    COUNT(DISTINCT point_report_month) AS point_months_cnt
FROM vd_month_upto_selected_metrics
WHERE report_month = '2026-05'
GROUP BY report_month;

B) Consistency check by month (activity totals vs cohort totals):

SELECT
    point_report_month,
    MAX(active_clients_cnt) + MAX(passive_clients_cnt) AS total_clients_by_activity,
    MAX(cohort_0_2500_cnt) + MAX(cohort_gt_2500_cnt) + MAX(cohort_lt_0_cnt) + MAX(cohort_eq_0_cnt) AS total_clients_by_cohorts,
    MAX(trx_sum_total) AS trx_sum_total
FROM vd_month_upto_selected_metrics
WHERE report_month = '2026-05'
GROUP BY point_report_month
ORDER BY point_report_month;
